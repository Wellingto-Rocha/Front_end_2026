import React, { StrictMode, useState } from "https://esm.sh/react";
import { createRoot } from "https://esm.sh/react-dom/client";

function Game() {
  const numeroSecreto = 7;

  const [palpite, setPalpite] = useState("");
  const [mensagem, setMensagem] = useState("");

  function verificarPalpite() {
    if (palpite.trim() === "") {
      setMensagem("Digite um número.");
      return;
    }

    const numero = Number(palpite);

    if (numero === numeroSecreto) {
      setMensagem("🎉 Parabéns! Você acertou!");
    } else if (numero < numeroSecreto) {
      setMensagem("📈 O número secreto é maior.");
    } else {
      setMensagem("📉 O número secreto é menor.");
    }
  }

  return (
    <div className="game">
      <h1>Jogo de Adivinhação</h1>

      <input
        type="number"
        value={palpite}
        onChange={(e) => setPalpite(e.target.value)}
        placeholder="Digite um número"
      />

      <button onClick={verificarPalpite}>
        Verificar
      </button>

      <div className="mensagem">
        {mensagem}
      </div>
    </div>
  );
}

const root = createRoot(document.getElementById("root"));

root.render(
  <StrictMode>
    <Game />
  </StrictMode>
);
